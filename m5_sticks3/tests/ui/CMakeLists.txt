cmake_minimum_required(VERSION 3.16)
project(m5_peripheral_ui_tests LANGUAGES C CXX)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(peripheral_ui_tests ../peripheral_ui_tests.cpp ../../components/app_modules/src/peripheral_ui.cpp)
target_compile_features(peripheral_ui_tests PRIVATE cxx_std_20)
target_compile_options(peripheral_ui_tests PRIVATE -Wall -Wextra -Werror)
target_include_directories(peripheral_ui_tests PRIVATE ../ui_stubs ../../components/audio/include
    ../../components/infrared/include ../../components/app_modules/src)
target_link_libraries(peripheral_ui_tests PRIVATE lvgl m)
enable_testing()
add_test(NAME peripheral_ui_tests COMMAND peripheral_ui_tests)
